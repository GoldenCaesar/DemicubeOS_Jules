<script data-purpose="dump-simulation">
    // Simulates realistic memory dumping progress ticker
    (function () {
      let percent = 84;
      const pctElement = document.getElementById('dump-pct');
      const progressBar = document.querySelector('[data-purpose="progress-fill"]');

      if (!pctElement || !progressBar) return;

      const interval = setInterval(() => {
        if (percent < 99) {
          percent += 1;
          pctElement.textContent = percent + '% Complete';
          progressBar.style.width = percent + '%';
        } else if (percent === 99) {
          percent = 100;
          pctElement.textContent = '100% Core Dump Complete. System Halted.';
          progressBar.style.width = '100%';
          clearInterval(interval);
        }
      }, 700);

      // Keyboard handler for simulated emergency keys
      window.addEventListener('keydown', (e) => {
        if (['F1', 'F2', 'F8', 'Escape'].includes(e.key)) {
          e.preventDefault();
          const origTitle = document.title;
          document.title = `[EXECUTING ACTION: ${e.key}]`;
          setTimeout(() => {
            document.title = origTitle;
          }, 1200);
        }
      });
    })();
  </script>
